   156→  /** Modell wechseln (oder gleiches neu starten): disposed alles Alte. */
   157→  async loadModel(id: ModelId): Promise<void> {
   158→    if (this.loading) return;
   159→    this.loading = true;
   160→    this.loadingText = id === "microduck" ? "Lade Ente…" : "Lade G1 (29 Gelenke)…";
   161→    this.emit();
   162→    try {
   163→      // Training anhalten + Trainer wegwerfen (Obs-Dimensionen ändern sich!)
   164→      this.trainingActive = false;
   165→      this.trainer?.dispose();
   166→      this.trainer = null;
   167→      this.esView = null;
   168→      this.resetTransientState();
   169→      this.mode = "manuell";
   170→
   171→      const meta = getModel(id);
   172→      const needRig = this.modelId !== id ||
   173→        !(this.world && (id === "microduck" ? this.world.duckRig : this.world.g1Rig));
   174→
   175→      // Welt-Konfiguration (global persistiert) auf dieses Modell anwenden
   176→      this.worldCfg = loadWorldPrefs();
   177→      this.worldBuild = this.worldCfg.enabled ? generateWorld(this.worldCfg, id) : null;
   178→      this.applyWorldVisuals();
   179→      this.pointMode = loadPointPrefs(id);
   180→      this.point = [0.8, 0];
   181→
   182→      const tasks: Promise<any>[] = [this.engine.load(id, this.worldBuild)];
   183→      if (needRig && this.world) tasks.push(mountRig(this.world, id));
   184→      if (id === "microduck") tasks.push(this.getOnnxSession(meta.defaultOnnx).catch(() => null));
   185→      await Promise.all(tasks);
   186→
   187→      this.modelId = id;
   188→      this.selectedOnnx = meta.defaultOnnx;
   189→      this.source = "onnx";
   190→      this.g1Base = new Float32Array(this.engine.standPose);
   191→      this.esObsExtra = 0;
   192→      this.engine.setImitObs(false);
   193→      this.engine.targetPoint = this.pointMode ? [...this.point] as [number, number] : null;
   194→      this.prevAct = new Float32Array(meta.actionDim);
   195→      this.actBuf = new Float32Array(meta.actionDim);
   196→      // Mappings + Bewertung pro Modell laden (Persistenz)
   197→      this.mappings = loadMapping(id);
   198→      this.rewardCfg = loadRewardConfig(id);
   199→      this.testUptime = 0;
   200→      this.testReward = 0;
   201→      this.error = null;
   202→    } catch (err: any) {
   203→      this.error = err?.message || String(err);
   204→      console.error("[core] Modell-Load fehlgeschlagen:", err);
   205→    } finally {
   206→      this.loading = false;
   207→      this.emit();
   208→    }

   338→  /** GLB-Clip bauen und aktivieren; Reward-Terme automatisch scharf schalten. */
   339→  async activateAnimation(
   340→    buffer: ArrayBuffer, clipIndex: number, mirror: boolean, modelId: ModelId,
   341→  ): Promise<{ ok: boolean; message: string }> {
   342→    try {
   343→      if (this.trainingActive) this.stopTraining();
   344→      this.trainer?.dispose();
   345→      this.trainer = null;
   346→      this.esView = null;
   347→      this.esObsExtra = 0;
   348→      const meta = getModel(modelId);
   349→      const clip = await buildImitClip(buffer, clipIndex, meta, mirror);
   350→      if (clip.mapped === 0) {
   351→        return { ok: false, message: "Keine Gelenke gemappt – Skeleton-Namen nicht erkannt." };
   352→      }
   353→      this.imitClip = clip;
   354→      this.imitPlaying = true;
   355→      this.imitTime = 0;
   356→      this.imitTargetBuf = new Float32Array(clip.dim);
   357→      this.engine.imitTarget = this.imitTargetBuf;
   358→      this.engine.imitRootDelta = null;
   359→      this.engine.setImitObs(true);
   360→      saveImitPrefs(modelId, clip.name, mirror);
   361→      // Terme automatisch aktivieren, falls beide aus
   362→      if (this.rewardCfg) {
   363→        const im = this.rewardCfg.terms.imitate;
   364→        const ih = this.rewardCfg.terms.imitHeight;
   365→        let changed = false;
   366→        if (im && !im.enabled) { im.enabled = true; im.weight = Math.max(1.5, Math.abs(im.weight)); changed = true; }
   367→        if (ih && !ih.enabled) { ih.enabled = true; ih.weight = Math.max(1.0, Math.abs(ih.weight)); changed = true; }
   368→        if (changed && this.modelId) saveRewardConfig(this.modelId, this.rewardCfg);
   369→      }
   370→      this.emit();
   371→      return {
   372→        ok: true,
   373→        message: `${clip.mapped} Gelenke gemappt · ${clip.duration.toFixed(1)} s · ${clip.name}`,
   374→      };
   375→    } catch (err: any) {
   376→      console.error("[core] Animations-Load fehlgeschlagen:", err);
   377→      return { ok: false, message: err?.message || String(err) };
   378→    }
   379→  }

   440→  private getOnnxSession(id: string): Promise<OnnxPolicy | null> {
   441→    const meta = this.modelId ? getModel(this.modelId) : null;
   442→    const def = meta?.policies.find((p) => p.id === id);
   443→    if (!def) return Promise.resolve(null);
   444→    let p = this.sessions.get(def.url);
   445→    if (!p) {
   446→      p = OnnxPolicy.create(def.url).catch((err) => {
   447→        console.warn("[core] ONNX-Load fehlgeschlagen:", def.url, err);
   448→        this.sessions.delete(def.url);
   449→        throw err;
   450→      });
   451→      this.sessions.set(def.url, p);
   452→    }
   453→    return p;
   454→  }

   456→  // ── ES-Training ─────────────────────────────────────────────────────────────

   458→  async startTraining(): Promise<void> {
   459→    if (this.trainingActive || !this.modelId) return;
   460→    try {
   461→      const imitActive = !!(this.imitClip && this.imitPlaying);
   462→      const obsExtra = imitActive ? 2 : 0;
   463→      const imitData: ImitEvalData | null = imitActive && this.imitClip
   464→        ? {
   465→            dim: this.imitClip.dim, frames: this.imitClip.frames, fps: this.imitClip.fps,
   466→            duration: this.imitClip.duration, targets: this.imitClip.targets,
   467→            rootY: this.imitClip.rootY, baseY: this.imitClip.baseY,
   468→          }
   469→        : null;
   470→      const imitSig = this.imitClip
   471→        ? `${this.imitClip.name}|${this.imitClip.frames}|${this.imitClip.mapped}`
   472→        : "";
   473→      // Trainer wegwerfen, wenn Welt/Imitation/Obs-Layout sich geändert haben
   474→      if (this.trainer) {
   475→        const stale =
   476→          this.trainer.obsExtra !== obsExtra ||
   477→          this.trainer.world !== this.worldBuild ||
   478→          this.trainer.imitSig !== imitSig;
   479→        if (stale) {
   480→          this.trainer.dispose();
   481→          this.trainer = null;
   482→          this.esView = null;
   483→          this.esObsExtra = 0;
   484→        }
   485→      }
   486→      if (!this.trainer) {
   487→        const warm = await warmStartMlp(getModel(this.modelId), obsExtra);
   488→        this.trainer = new EsTrainer(this.modelId, warm.theta, warm.layout, this.rewardCfg!, {
   489→          obsExtra, world: this.worldBuild, imit: imitData, imitSig,
   490→        });
   491→        await this.trainer.init();
   492→      }
   493→      this.trainer.setReward(this.rewardCfg!);
   494→      this.trainer.setTargetPoint(this.pointMode ? ([...this.point] as [number, number]) : null);
   495→      this.trainingActive = true;
   496→      this.emit();
   497→      await this.trainer.tryStartWorkers();
   498→      if (!this.trainingActive) return; // inzwischen gestoppt
   499→      void this.trainingLoop();
   500→    } catch (err: any) {
   501→      console.error("[core] Training-Start fehlgeschlagen:", err);
   502→      this.trainingActive = false;
   503→      this.emit();
   504→    }
   505→  }
   506→

   508→    this.trainingActive = false;
   509→    this.emit();
   510→  }

   517→  private async trainingLoop() {
   518→    while (this.trainingActive && this.trainer) {
   519→      await this.trainer.runGeneration();
   520→      this.updateEsView();
   521→      this.emit();
   522→      await new Promise((r) => setTimeout(r, 30));
   523→    }
   524→  }
   525→
   526→  private updateEsView() {
   527→    if (!this.trainer) {
   528→      this.esView = null;
   529→      return;
   530→    }
   531→    try {
   532→      this.esView = new MlpPolicy(this.trainer.layout, this.trainer.bestTheta);
   533→      this.esObsExtra = this.trainer.layout.obsDim - this.trainer.meta.obsDim;
   534→    } catch {
   535→      this.esView = null;
   536→    }
   537→  }

   552→  async saveTheta(): Promise<boolean> {
   553→    if (!this.trainer || !this.modelId) return false;
   554→    try {
   555→      const t = this.trainer;
   556→      localStorage.setItem(this.thetaKey(), JSON.stringify({
   557→        layout: t.layout,
   558→        theta: Array.from(t.theta),
   559→        bestTheta: Array.from(t.bestTheta),
   560→        generation: t.generation,
   561→        bestEver: t.bestEver,
   562→        history: t.history.slice(-200),
   563→      }));
   564→      return true;
   565→    } catch {
   566→      return false;
   567→    }
   568→  }
   569→
   570→  hasSavedTheta(): boolean {
   571→    try {
   572→      return !!localStorage.getItem(this.thetaKey());
   573→    } catch {
   574→      return false;
   575→    }
   576→  }
   577→
   578→  async loadTheta(): Promise<boolean> {
   579→    if (!this.modelId) return false;
   580→    try {
   581→      const raw = localStorage.getItem(this.thetaKey());
   582→      if (!raw) return false;
   583→      const d = JSON.parse(raw) as {
   584→        layout: MlpLayout; theta: number[]; bestTheta: number[];
   585→        generation: number; bestEver: number; history: number[];
   586→      };
   587→      if (!d?.theta?.length || !d?.layout) return false;
   588→      this.trainingActive = false;
   589→      this.trainer?.dispose();
   590→      this.trainer = new EsTrainer(this.modelId, new Float32Array(d.theta), d.layout, this.rewardCfg!);
   591→      await this.trainer.init();
   592→      this.trainer.bestTheta = new Float32Array(d.bestTheta ?? d.theta);
   593→      this.trainer.generation = d.generation ?? 0;
   594→      this.trainer.bestEver = d.bestEver ?? -Infinity;
   595→      this.trainer.history = Array.isArray(d.history) ? d.history : [];
   596→      this.updateEsView();
   597→      this.emit();
   598→      return true;
   599→    } catch (err) {
   600→      console.warn("[core] Theta-Load fehlgeschlagen:", err);
   601→      return false;
   602→    }
   603→  }

   605→  // ── Kontroll-Loop (50 Hz, async wegen ONNX) ─────────────────────────────────
   606→
   607→  private stepMs(): number {
   608→    const m = this.engine.meta;
   609→    return m ? m.timestep * m.decimation * 1000 : 20;
   610→  }
   611→
   612→  private async controlLoop() {
   613→    let next = performance.now();
   614→    let count = 0;
   615→    let hzT0 = next;
   616→    while (this.running) {
   617→      try {
   618→        await this.controlStep();
   619→      } catch (err) {
   620→        console.warn("[core] controlStep:", err);
   621→      }
   622→      count++;
   623→      const now = performance.now();
   624→      if (now - hzT0 > 500) {
   625→        this.ctrlHz = (count * 1000) / (now - hzT0);
   626→        count = 0;
   627→        hzT0 = now;
   628→      }
   629→      next += this.stepMs();
   630→      const wait = next - performance.now();
   631→      if (wait > 0) await new Promise((r) => setTimeout(r, wait));
   632→      else next = performance.now(); // zurückgefallen: nicht spiralen
   633→    }
   634→  }
   635→
   636→  private async controlStep(): Promise<void> {
   637→    const engine = this.engine;
   638→    if (!engine.data || !engine.meta || !this.modelId) return;
   639→
   640→    // ── v2.1: Punkt-Modus – Joystick bewegt den 3D-Punkt ──
   641→    const dtStep = this.stepMs() / 1000;
   642→    if (this.pointMode) {
   643→      const sp = 1.4 * dtStep;
   644→      const lim = arenaHalf(this.modelId) - 0.15;
   645→      this.point[0] = Math.min(lim, Math.max(-lim, this.point[0] + this.joy.x * sp));
   646→      this.point[1] = Math.min(lim, Math.max(-lim, this.point[1] + this.joy.y * sp));
   647→      engine.targetPoint = [this.point[0], this.point[1]];
   648→      this.trainer?.setTargetPoint([this.point[0], this.point[1]]);
   649→      this.world?.setTargetPoint(this.point[0], this.point[1], true);
   650→    } else if (engine.targetPoint) {
   651→      engine.targetPoint = null;
   652→      this.world?.setTargetPoint(0, 0, false);
   653→    }
   654→
   655→    // ── v2.1: Imitation – Zielpose je Policy-Step ──
   656→    if (this.imitClip && this.imitPlaying && this.imitTargetBuf) {
   657→      this.imitTime += dtStep;
   658→      targetAt(this.imitClip, this.imitTime, this.imitTargetBuf);
   659→      engine.imitTarget = this.imitTargetBuf;
   660→      engine.imitRootDelta = rootDeltaAt(this.imitClip, this.imitTime);
   661→      const ph = this.imitClip.duration > 0
   662→        ? ((this.imitTime % this.imitClip.duration) / this.imitClip.duration) * 2 * Math.PI
   663→        : 0;
   664→      engine.imitPhase = [Math.sin(ph), Math.cos(ph)];
   665→    }
   666→
   667→    if (this.recovering) {
   668→      this.stepRecovery();
   669→    } else if (this.modelId === "microduck") {
   670→      await this.stepDuck();
   671→    } else {
   672→      this.stepG1();
   673→    }
   674→    if (!engine.data) return; // Modellwechsel während des Steps (dispose)
   675→
   676→    // Auto-Recovery nur im Test-Modus (Zustand läuft physisch weiter)
   677→    if (this.mode === "test" && this.autoRecovery && !this.recovering && engine.isFallen()) {
   678→      if (++this.fallDebounce >= 10) this.startRecovery();
   679→    } else if (!this.recovering) {
   680→      this.fallDebounce = 0;
   681→    }
   682→
   683→    if (this.mode === "test") {
   684→      this.testUptime = (performance.now() - this.testStart) / 1000;
   685→    }
   686→  }
   687→
   688→  // ── Enten-Step (cmd + ONNX/ES-Policy) ───────────────────────────────────────
   689→
   690→  private async stepDuck(): Promise<void> {
   691→    const engine = this.engine;
   692→    const meta = engine.meta;
   693→
   694→    // Command-Slots: Punkt-Modus → Richtung zum Punkt; sonst Mappings/Test
   695→    if (this.pointMode && engine.targetPoint) {
   696→      const [tx, ty] = engine.targetPoint;
   697→      const p = engine.torsoPos();
   698→      const dx = tx - p[0], dy = ty - p[1];
   699→      const dist = Math.hypot(dx, dy);
   700→      const yaw = engine.yaw();
   701→      const fwd = Math.cos(yaw) * dx + Math.sin(yaw) * dy;
   702→      const lat = -Math.sin(yaw) * dx + Math.cos(yaw) * dy;
   703→      const desired = Math.atan2(dy, dx);
   704→      let yerr = desired - yaw;
   705→      while (yerr > Math.PI) yerr -= 2 * Math.PI;
   706→      while (yerr < -Math.PI) yerr += 2 * Math.PI;
   707→      if (dist > 0.12) {
   708→        engine.cmd[0] = clamp(VEL_FWD * 0.9 * Math.max(0, Math.cos(yerr)), VEL_BACK, VEL_FWD);
   709→        engine.cmd[1] = clamp(lat * 1.2, -VEL_LAT, VEL_LAT);
   710→        engine.cmd[2] = clamp(1.4 * yerr, -VEL_ANG, VEL_ANG);
   711→      } else {
   712→        engine.cmd[0] = 0; engine.cmd[1] = 0; engine.cmd[2] = 0;
   713→      }
   714→    } else {
   715→      let vx = 0, vy = 0, wz = 0;
   716→      if (this.mode !== "test") {
   717→        for (const m of this.mappings) {
   718→          if (m.targetType !== "cmd") continue;
   719→          const v = (m.source === "joyX" ? this.joy.x : this.joy.y) * m.gain;
   720→          if (m.target === "cmd_x") vx += v;
   721→          else if (m.target === "cmd_y") vy += v;
   722→          else if (m.target === "cmd_yaw") wz -= v; // Stick rechts = Rechtsdrehung
   723→        }
   724→      }
   725→      engine.cmd[0] = clamp(vx, VEL_BACK, VEL_FWD);
   726→      engine.cmd[1] = clamp(vy, -VEL_LAT, VEL_LAT);
   727→      engine.cmd[2] = clamp(wz, -VEL_ANG, VEL_ANG);
   728→    }
   729→
   730→    // Button-Trigger (Edge)
   731→    if (this.pendingPolicyId) {
   732→      this.overridePolicy = this.pendingPolicyId;
   733→      this.pendingPolicyId = null;
   734→    }
   735→    if (this.pendingPoseId) {
   736→      const pose = poseById("microduck", this.pendingPoseId);
   737→      this.pendingPoseId = null;
   738→      if (pose) this.startDuckPose(pose.targets);
   739→    }
   740→
   741→    // v2.1: Imitation manuell anwenden (Zielpose direkt als ctrl)
   742→    if (this.mode === "manuell" && this.imitManual && this.imitClip && this.imitPlaying && this.imitTargetBuf) {
   743→      engine.applyCtrlFromPose(this.imitTargetBuf);
   744→      engine.stepPhysics();
   745→      return;
   746→    }
   747→
   748→    // Pose-Hold (kurze Overlay-Phase, dann zurück zur Policy)
   749→    if (this.duckPoseHold) {
   750→      const hold = this.duckPoseHold;
   751→      const now = performance.now();
   752→      const u = clamp((now - hold.t0) / hold.blendMs, 0, 1);
   753→      const ctrl = engine.data.ctrl as Float32Array;
   754→      for (let j = 0; j < meta.actionDim; j++) {
   755→        ctrl[j] = hold.from[j] + (hold.to[j] - hold.from[j]) * u;
   756→      }
   757→      engine.stepPhysics();
   758→      if (now - hold.t0 > hold.blendMs + hold.holdMs) {
   759→        this.duckPoseHold = null;
   760→        engine.lastAction.fill(0);
   761→      }
   762→      return;
   763→    }
   764→
   765→    // Policy wählen
   766→    let policy: OnnxPolicy | MlpPolicy | null = null;
   767→    if (this.source === "es" && this.esView) {
   768→      policy = this.esView;
   769→    } else {
   770→      const id = this.overridePolicy ?? this.selectedOnnx;
   771→      policy = (await this.getOnnxSession(id)) ?? (await this.getOnnxSession(meta.defaultOnnx));
   772→      if (!engine.data) return; // Modellwechsel während des Session-Ladens
   773→    }
   774→    if (!policy) {
   775→      engine.stepPhysics();
   776→      return;
   777→    }
   778→
   779→    // ONNX braucht exakte Basis-Dimension; ES-MLP ggf. +2 (Imitations-Phase)
   780→    const obs = policy instanceof MlpPolicy
   781→      ? engine.obsFor(this.esObsExtra >= 2)
   782→      : engine.obsFor(false);
   783→    const act = policy instanceof MlpPolicy
   784→      ? policy.forward(obs, this.actBuf)
   785→      : await policy.forward(obs);
   786→    this.actBuf.fill(0);
   787→    this.actBuf.set(act.subarray(0, Math.min(act.length, this.actBuf.length)));
   788→    engine.lastAction.set(this.actBuf);
   789→    engine.stepWithAction(this.actBuf);
   790→
   791→    if (this.mode === "test") {
   792→      this.testReward += stepRewardValue(engine, this.rewardCfg!, this.actBuf, this.prevAct);
   793→      this.prevAct.set(this.actBuf);
   794→    }
   795→  }
   796→
   797→  private startDuckPose(targets: Record<string, number>) {
   798→    const meta = this.engine.meta;
   799→    const from = new Float32Array(this.engine.data.ctrl as Float32Array);
   800→    const to = new Float32Array(meta.actionDim);
   801→    for (let j = 0; j < meta.actionDim; j++) {
   802→      const name = meta.jointNames[j];
   803→      to[j] = targets[name] ?? meta.defaultPose[j];
   804→    }
   805→    this.duckPoseHold = { from, to, t0: performance.now(), blendMs: 400, holdMs: 1600 };
   806→  }
   807→
   808→  // ── G1-Step (direkte Gelenk-Targets / ES-Policy) ────────────────────────────
   809→
   810→  private stepG1(): void {
   811→    const engine = this.engine;
   812→    const meta = engine.meta;
   813→    const dim = meta.actionDim;
   814→
   815→    // Pose-Trigger (Edge)
   816→    if (this.pendingPoseId) {
   817→      const pose = poseById("unitree_g1", this.pendingPoseId);
   818→      this.pendingPoseId = null;
   819→      if (pose) this.startG1Blend(pose.targets);
   820→    }
   821→
   822→    // Basis-Blending (0.4 s)
   823→    if (this.g1Blend && this.g1Base) {
   824→      const b = this.g1Blend;
   825→      const u = clamp((performance.now() - b.t0) / b.dur, 0, 1);
   826→      const e = u < 0.5 ? 2 * u * u : 1 - Math.pow(-2 * u + 2, 2) / 2;
   827→      for (let j = 0; j < dim; j++) this.g1Base[j] = b.from[j] + (b.to[j] - b.from[j]) * e;
   828→      if (u >= 1) this.g1Blend = null;
   829→    }
   830→
   831→    if (this.mode === "manuell") {
   832→      // Joystick-Offsets auf gemappten Gelenken
   833→      const targets = new Float32Array(dim);
   834→      targets.set(this.g1Base ?? engine.standPose);
   835→      for (const m of this.mappings) {
   836→        if (m.targetType !== "gelenk") continue;
   837→        const v = (m.source === "joyX" ? this.joy.x : this.joy.y) * m.gain;
   838→        for (const jn of m.target.split(",").map((s) => s.trim())) {
   839→          const idx = meta.jointNames.indexOf(jn);
   840→          if (idx >= 0) targets[idx] += v;
   841→        }
   842→      }
   843→      engine.applyCtrlFromPose(targets);
   844→      engine.stepPhysics();
   845→      return;
   846→    }
   847→
   848→    // v2.1: Punkt-Modus – cmd Richtung Punkt (für ES-Obs)
   849→    if (this.pointMode && engine.targetPoint) {
   850→      const [tx, ty] = engine.targetPoint;
   851→      const p = engine.torsoPos();
   852→      const dx = tx - p[0], dy = ty - p[1];
   853→      const dist = Math.hypot(dx, dy);
   854→      const yaw = engine.yaw();
   855→      const desired = Math.atan2(dy, dx);
   856→      let yerr = desired - yaw;
   857→      while (yerr > Math.PI) yerr -= 2 * Math.PI;
   858→      while (yerr < -Math.PI) yerr += 2 * Math.PI;
   859→      engine.cmd[0] = dist > 0.15 ? 0.35 * Math.max(0, Math.cos(yerr)) : 0;
   860→      engine.cmd[2] = clamp(1.2 * yerr, -1, 1);
   861→    }
   862→
   863→    // Training/Test: ES-Policy (G1 hat keine ONNX-Policies)
   864→    const mlp = this.esView;
   865→    if (!mlp || !this.g1Base) {
   866→      // noch nichts trainiert: Stand-Pose halten, Physik läuft weiter
   867→      engine.applyCtrlFromPose(engine.standPose);
   868→      engine.stepPhysics();
   869→      return;
   870→    }
   871→    const obs = engine.obsFor(this.esObsExtra >= 2);
   872→    const act = mlp.forward(obs, this.actBuf);
   873→    engine.lastAction.set(act);
   874→    engine.stepWithAction(act);
   875→    if (this.mode === "test") {
   876→      this.testReward += stepRewardValue(engine, this.rewardCfg!, act, this.prevAct);
   877→      this.prevAct.set(act);
   878→    }
   879→  }
   880→
   881→  private startG1Blend(targets: Record<string, number>) {
   882→    const meta = this.engine.meta;
   883→    const to = new Float32Array(meta.actionDim);
   884→    for (let j = 0; j < meta.actionDim; j++) {
   885→      to[j] = targets[meta.jointNames[j]] ?? this.engine.standPose[j];
   886→    }
   887→    this.g1Blend = { from: (this.g1Base ?? this.engine.standPose).slice(), to, t0: performance.now(), dur: 400 };
   888→  }

   890→  // ── Auto-Recovery (sanftes Blenden zur Stand-Pose, kein Teleport) ───────────
   891→
   892→  private startRecovery() {
   893→    this.recovering = true;
   894→    this.recoverySteps = 0;
   895→    this.recoveryUpright = 0;
   896→    this.fallDebounce = 0;
   897→    this.engine.lastAction.fill(0);
   898→    this.emit();
   899→  }
   900→
   901→  private stepRecovery() {
   902→    const engine = this.engine;
   903→    const meta = engine.meta;
   904→    const stand = meta.id === "microduck" ? Float32Array.from(meta.defaultPose) : engine.standPose;
   905→    const ctrl = engine.data.ctrl as Float32Array;
   906→    for (let j = 0; j < meta.actionDim; j++) {
   907→      ctrl[j] += (stand[j] - ctrl[j]) * 0.06; // sanft, pro Policy-Step
   908→    }
   909→    engine.stepPhysics();
   910→    this.recoverySteps++;
   911→    const upright = engine.projGravZ() < -0.85 && !engine.isFallen();
   912→    this.recoveryUpright = upright ? this.recoveryUpright + 1 : 0;
   913→    if (this.recoveryUpright >= 50) {
   914→      // 1 s aufrecht → Policy übernimmt wieder (Zustand läuft weiter)
   915→      this.recovering = false;
   916→      this.recoveryUpright = 0;
   917→      engine.lastAction.fill(0);
   918→      this.prevAct.fill(0);
   919→      this.emit();
   920→    }
   921→  }

   925→  private renderLoop = () => {
   926→    if (!this.running || !this.world) return;
   927→    const now = performance.now();
   928→    const dt = Math.min(0.05, (now - this.lastFrameT) / 1000 || 0.016);
   929→    this.lastFrameT = now;
   930→    const engine = this.engine;
   931→    if (engine.data && engine.meta) {
   932→      const [x, y, z] = engine.torsoPos();
   933→      this.world.setCameraTarget(x, z, -y);
   934→      if (engine.meta.id === "microduck") {
   935→        syncDuck(this.world, engine);
   936→      } else {
   937→        syncG1(this.world, engine, engine.mujoco);
   938→      }
   939→      syncBall(this.world, engine);
   940→      this.world.updateCamera(dt);
   941→      this.world.renderer.render(this.world.scene, this.world.camera);
   942→    }
   943→    if (now - this.lastEmit > 250) {
   944→      this.lastEmit = now;
   945→      this.emit();
   946→    }
   947→    requestAnimationFrame(this.renderLoop);
   948→  };

  1001→  static async toggleFullscreen(el: HTMLElement): Promise<void> {
  1002→    const doc = document as Document & {
  1003→      webkitFullscreenElement?: Element;
  1004→      webkitExitFullscreen?: () => Promise<void>;
  1005→    };
  1006→    const full = el as HTMLElement & { webkitRequestFullscreen?: () => Promise<void> };
  1007→    const active = document.fullscreenElement ?? doc.webkitFullscreenElement;
  1008→    try {
  1009→      if (active) {
  1010→        if (document.exitFullscreen) await document.exitFullscreen();
  1011→        else await doc.webkitExitFullscreen?.();
  1012→        return;
  1013→      }
  1014→      if (el.requestFullscreen) await el.requestFullscreen();
  1015→      else if (full.webkitRequestFullscreen) await full.webkitRequestFullscreen();
  1016→      const orientation = screen.orientation as ScreenOrientation & {
  1017→        lock?: (o: string) => Promise<void>;
  1018→      };
  1019→      await orientation?.lock?.("landscape")?.catch?.(() => {});
  1020→    } catch {
  1021→      // Vollbild/Rotation verweigert (z. B. iOS Safari) – ignorieren
  1022→    }
  1023→  }

  1041→function loadWorldPrefs(): WorldConfig {
  1042→  try {
  1043→    const raw = localStorage.getItem("mdt_v2_world");
  1044→    if (raw) {
