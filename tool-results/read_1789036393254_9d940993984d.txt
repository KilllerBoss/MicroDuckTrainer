   156→   156→  /** Modell wechseln (oder gleiches neu starten): disposed alles Alte. */
   157→   157→  async loadModel(id: ModelId): Promise<void> {
   158→   158→    if (this.loading) return;
   159→   159→    this.loading = true;
   160→   160→    this.loadingText = id === "microduck" ? "Lade Ente…" : "Lade G1 (29 Gelenke)…";
   161→   161→    this.emit();
   162→   162→    try {
   163→   163→      // Training anhalten + Trainer wegwerfen (Obs-Dimensionen ändern sich!)
   164→   164→      this.trainingActive = false;
   165→   165→      this.trainer?.dispose();
   166→   166→      this.trainer = null;
   167→   167→      this.esView = null;
   168→   168→      this.resetTransientState();
   169→   169→      this.mode = "manuell";
   170→   170→
   171→   171→      const meta = getModel(id);
   172→   172→      const needRig = this.modelId !== id ||
   173→   173→        !(this.world && (id === "microduck" ? this.world.duckRig : this.world.g1Rig));
   174→   174→
   175→   175→      // Welt-Konfiguration (global persistiert) auf dieses Modell anwenden
   176→   176→      this.worldCfg = loadWorldPrefs();
   177→   177→      this.worldBuild = this.worldCfg.enabled ? generateWorld(this.worldCfg, id) : null;
   178→   178→      this.applyWorldVisuals();
   179→   179→      this.pointMode = loadPointPrefs(id);
   180→   180→      this.point = [0.8, 0];
   181→   181→
   182→   182→      const tasks: Promise<any>[] = [this.engine.load(id, this.worldBuild)];
   183→   183→      if (needRig && this.world) tasks.push(mountRig(this.world, id));
   184→   184→      if (id === "microduck") tasks.push(this.getOnnxSession(meta.defaultOnnx).catch(() => null));
   185→   185→      await Promise.all(tasks);
   186→   186→
   187→   187→      this.modelId = id;
   188→   188→      this.selectedOnnx = meta.defaultOnnx;
   189→   189→      this.source = "onnx";
   190→   190→      this.g1Base = new Float32Array(this.engine.standPose);
   191→   191→      this.esObsExtra = 0;
   192→   192→      this.engine.setImitObs(false);
   193→   193→      this.engine.targetPoint = this.pointMode ? [...this.point] as [number, number] : null;
   194→   194→      this.prevAct = new Float32Array(meta.actionDim);
   195→   195→      this.actBuf = new Float32Array(meta.actionDim);
   196→   196→      // Mappings + Bewertung pro Modell laden (Persistenz)
   197→   197→      this.mappings = loadMapping(id);
   198→   198→      this.rewardCfg = loadRewardConfig(id);
   199→   199→      this.testUptime = 0;
   200→   200→      this.testReward = 0;
   201→   201→      this.error = null;
   202→   202→    } catch (err: any) {
   203→   203→      this.error = err?.message || String(err);
   204→   204→      console.error("[core] Modell-Load fehlgeschlagen:", err);
   205→   205→    } finally {
   206→   206→      this.loading = false;
   207→   207→      this.emit();
   208→   208→    }
   209→   209→  }

   338→   338→  /** GLB-Clip bauen und aktivieren; Reward-Terme automatisch scharf schalten. */
   339→   339→  async activateAnimation(
   340→   340→    buffer: ArrayBuffer, clipIndex: number, mirror: boolean, modelId: ModelId,
   341→   341→  ): Promise<{ ok: boolean; message: string }> {
   342→   342→    try {
   343→   343→      if (this.trainingActive) this.stopTraining();
   344→   344→      this.trainer?.dispose();
   345→   345→      this.trainer = null;
   346→   346→      this.esView = null;
   347→   347→      this.esObsExtra = 0;
   348→   348→      const meta = getModel(modelId);
   349→   349→      const clip = await buildImitClip(buffer, clipIndex, meta, mirror);
   350→   350→      if (clip.mapped === 0) {
   351→   351→        return { ok: false, message: "Keine Gelenke gemappt – Skeleton-Namen nicht erkannt." };
   352→   352→      }
   353→   353→      this.imitClip = clip;
   354→   354→      this.imitPlaying = true;
   355→   355→      this.imitTime = 0;
   356→   356→      this.imitTargetBuf = new Float32Array(clip.dim);
   357→   357→      this.engine.imitTarget = this.imitTargetBuf;
   358→   358→      this.engine.imitRootDelta = null;
   359→   359→      this.engine.setImitObs(true);
   360→   360→      saveImitPrefs(modelId, clip.name, mirror);
   361→   361→      // Terme automatisch aktivieren, falls beide aus
   362→   362→      if (this.rewardCfg) {
   363→   363→        const im = this.rewardCfg.terms.imitate;
   364→   364→        const ih = this.rewardCfg.terms.imitHeight;
   365→   365→        let changed = false;
   366→   366→        if (im && !im.enabled) { im.enabled = true; im.weight = Math.max(1.5, Math.abs(im.weight)); changed = true; }
   367→   367→        if (ih && !ih.enabled) { ih.enabled = true; ih.weight = Math.max(1.0, Math.abs(ih.weight)); changed = true; }
   368→   368→        if (changed && this.modelId) saveRewardConfig(this.modelId, this.rewardCfg);
   369→   369→      }
   370→   370→      this.emit();
   371→   371→      return {
   372→   372→        ok: true,
   373→   373→        message: `${clip.mapped} Gelenke gemappt · ${clip.duration.toFixed(1)} s · ${clip.name}`,
   374→   374→      };
   375→   375→    } catch (err: any) {
   376→   376→      console.error("[core] Animations-Load fehlgeschlagen:", err);
   377→   377→      return { ok: false, message: err?.message || String(err) };
   378→   378→    }
   379→   379→  }

   440→   440→  private getOnnxSession(id: string): Promise<OnnxPolicy | null> {
   441→   441→    const meta = this.modelId ? getModel(this.modelId) : null;
   442→   442→    const def = meta?.policies.find((p) => p.id === id);
   443→   443→    if (!def) return Promise.resolve(null);
   444→   444→    let p = this.sessions.get(def.url);
   445→   445→    if (!p) {
   446→   446→      p = OnnxPolicy.create(def.url).catch((err) => {
   447→   447→        console.warn("[core] ONNX-Load fehlgeschlagen:", def.url, err);
   448→   448→        this.sessions.delete(def.url);
   449→   449→        throw err;
   450→   450→      });
   451→   451→      this.sessions.set(def.url, p);
   452→   452→    }
   453→   453→    return p;
   454→   454→  }

   458→   458→  async startTraining(): Promise<void> {
   459→   459→    if (this.trainingActive || !this.modelId) return;
   460→   460→    try {
   461→   461→      const imitActive = !!(this.imitClip && this.imitPlaying);
   462→   462→      const obsExtra = imitActive ? 2 : 0;
   463→   463→      const imitData: ImitEvalData | null = imitActive && this.imitClip
   464→   464→        ? {
   465→   465→            dim: this.imitClip.dim, frames: this.imitClip.frames, fps: this.imitClip.fps,
   466→   466→            duration: this.imitClip.duration, targets: this.imitClip.targets,
   467→   467→            rootY: this.imitClip.rootY, baseY: this.imitClip.baseY,
   468→   468→          }
   469→   469→        : null;
   470→   470→      const imitSig = this.imitClip
   471→   471→        ? `${this.imitClip.name}|${this.imitClip.frames}|${this.imitClip.mapped}`
   472→   472→        : "";
   473→   473→      // Trainer wegwerfen, wenn Welt/Imitation/Obs-Layout sich geändert haben
   474→   474→      if (this.trainer) {
   475→   475→        const stale =
   476→   476→          this.trainer.obsExtra !== obsExtra ||
   477→   477→          this.trainer.world !== this.worldBuild ||
   478→   478→          this.trainer.imitSig !== imitSig;
   479→   479→        if (stale) {
   480→   480→          this.trainer.dispose();
   481→   481→          this.trainer = null;
   482→   482→          this.esView = null;
   483→   483→          this.esObsExtra = 0;
   484→   484→        }
   485→   485→      }
   486→   486→      if (!this.trainer) {
   487→   487→        const warm = await warmStartMlp(getModel(this.modelId), obsExtra);
   488→   488→        this.trainer = new EsTrainer(this.modelId, warm.theta, warm.layout, this.rewardCfg!, {
   489→   489→          obsExtra, world: this.worldBuild, imit: imitData, imitSig,
   490→   490→        });
   491→   491→        await this.trainer.init();
   492→   492→      }
   493→   493→      this.trainer.setReward(this.rewardCfg!);
   494→   494→      this.trainer.setTargetPoint(this.pointMode ? ([...this.point] as [number, number]) : null);
   495→   495→      this.trainingActive = true;
   496→   496→      this.emit();
   497→   497→      await this.trainer.tryStartWorkers();
   498→   498→      if (!this.trainingActive) return; // inzwischen gestoppt
   499→   499→      void this.trainingLoop();
   500→   500→    } catch (err: any) {
   501→   501→      console.error("[core] Training-Start fehlgeschlagen:", err);
   502→   502→      this.trainingActive = false;
   503→   503→      this.emit();
   504→   504→    }
   505→   505→  }

   510→   510→  }

   607→   607→  private stepMs(): number {
   608→   608→    const m = this.engine.meta;

   611→   611→
   612→   612→  private async controlLoop() {
   613→   613→    let next = performance.now();
   614→   614→    let count = 0;
   615→   615→    let hzT0 = next;
   616→   616→    while (this.running) {
   617→   617→      try {
   618→   618→        await this.controlStep();
   619→   619→      } catch (err) {
   620→   620→        console.warn("[core] controlStep:", err);
   621→   621→      }
   622→   622→      count++;
   623→   623→      const now = performance.now();
   624→   624→      if (now - hzT0 > 500) {
   625→   625→        this.ctrlHz = (count * 1000) / (now - hzT0);
   626→   626→        count = 0;
   627→   627→        hzT0 = now;
   628→   628→      }
   629→   629→      next += this.stepMs();
   630→   630→      const wait = next - performance.now();
   631→   631→      if (wait > 0) await new Promise((r) => setTimeout(r, wait));
   632→   632→      else next = performance.now(); // zurückgefallen: nicht spiralen
   633→   633→    }
   634→   634→  }
   635→   635→
   636→   636→  private async controlStep(): Promise<void> {
   637→   637→    const engine = this.engine;
   638→   638→    if (!engine.data || !engine.meta || !this.modelId) return;
   639→   639→
   640→   640→    // ── v2.1: Punkt-Modus – Joystick bewegt den 3D-Punkt ──
   641→   641→    const dtStep = this.stepMs() / 1000;
   642→   642→    if (this.pointMode) {
   643→   643→      const sp = 1.4 * dtStep;
   644→   644→      const lim = arenaHalf(this.modelId) - 0.15;
   645→   645→      this.point[0] = Math.min(lim, Math.max(-lim, this.point[0] + this.joy.x * sp));
   646→   646→      this.point[1] = Math.min(lim, Math.max(-lim, this.point[1] + this.joy.y * sp));
   647→   647→      engine.targetPoint = [this.point[0], this.point[1]];
   648→   648→      this.trainer?.setTargetPoint([this.point[0], this.point[1]]);
   649→   649→      this.world?.setTargetPoint(this.point[0], this.point[1], true);
   650→   650→    } else if (engine.targetPoint) {
   651→   651→      engine.targetPoint = null;
   652→   652→      this.world?.setTargetPoint(0, 0, false);
   653→   653→    }
   654→   654→
   655→   655→    // ── v2.1: Imitation – Zielpose je Policy-Step ──
   656→   656→    if (this.imitClip && this.imitPlaying && this.imitTargetBuf) {
   657→   657→      this.imitTime += dtStep;
   658→   658→      targetAt(this.imitClip, this.imitTime, this.imitTargetBuf);
   659→   659→      engine.imitTarget = this.imitTargetBuf;
   660→   660→      engine.imitRootDelta = rootDeltaAt(this.imitClip, this.imitTime);
   661→   661→      const ph = this.imitClip.duration > 0
   662→   662→        ? ((this.imitTime % this.imitClip.duration) / this.imitClip.duration) * 2 * Math.PI
   663→   663→        : 0;
   664→   664→      engine.imitPhase = [Math.sin(ph), Math.cos(ph)];
   665→   665→    }
   666→   666→
   667→   667→    if (this.recovering) {
   668→   668→      this.stepRecovery();
   669→   669→    } else if (this.modelId === "microduck") {
   670→   670→      await this.stepDuck();
   671→   671→    } else {
   672→   672→      this.stepG1();
   673→   673→    }
   674→   674→    if (!engine.data) return; // Modellwechsel während des Steps (dispose)
   675→   675→
   676→   676→    // Auto-Recovery nur im Test-Modus (Zustand läuft physisch weiter)
   677→   677→    if (this.mode === "test" && this.autoRecovery && !this.recovering && engine.isFallen()) {
   678→   678→      if (++this.fallDebounce >= 10) this.startRecovery();
   679→   679→    } else if (!this.recovering) {
   680→   680→      this.fallDebounce = 0;
   681→   681→    }
   682→   682→
   683→   683→    if (this.mode === "test") {
   684→   684→      this.testUptime = (performance.now() - this.testStart) / 1000;
   685→   685→    }
   686→   686→  }
   687→   687→
   688→   688→  // ── Enten-Step (cmd + ONNX/ES-Policy) ───────────────────────────────────────
   689→   689→
   690→   690→  private async stepDuck(): Promise<void> {
   691→   691→    const engine = this.engine;
   692→   692→    const meta = engine.meta;
   693→   693→
   694→   694→    // Command-Slots: Punkt-Modus → Richtung zum Punkt; sonst Mappings/Test
   695→   695→    if (this.pointMode && engine.targetPoint) {
   696→   696→      const [tx, ty] = engine.targetPoint;
   697→   697→      const p = engine.torsoPos();
   698→   698→      const dx = tx - p[0], dy = ty - p[1];
   699→   699→      const dist = Math.hypot(dx, dy);
   700→   700→      const yaw = engine.yaw();
   701→   701→      const fwd = Math.cos(yaw) * dx + Math.sin(yaw) * dy;
   702→   702→      const lat = -Math.sin(yaw) * dx + Math.cos(yaw) * dy;
   703→   703→      const desired = Math.atan2(dy, dx);
   704→   704→      let yerr = desired - yaw;
   705→   705→      while (yerr > Math.PI) yerr -= 2 * Math.PI;
   706→   706→      while (yerr < -Math.PI) yerr += 2 * Math.PI;
   707→   707→      if (dist > 0.12) {
   708→   708→        engine.cmd[0] = clamp(VEL_FWD * 0.9 * Math.max(0, Math.cos(yerr)), VEL_BACK, VEL_FWD);
   709→   709→        engine.cmd[1] = clamp(lat * 1.2, -VEL_LAT, VEL_LAT);
   710→   710→        engine.cmd[2] = clamp(1.4 * yerr, -VEL_ANG, VEL_ANG);
   711→   711→      } else {
   712→   712→        engine.cmd[0] = 0; engine.cmd[1] = 0; engine.cmd[2] = 0;
   713→   713→      }
   714→   714→    } else {
   715→   715→      let vx = 0, vy = 0, wz = 0;
   716→   716→      if (this.mode !== "test") {
   717→   717→        for (const m of this.mappings) {
   718→   718→          if (m.targetType !== "cmd") continue;
   719→   719→          const v = (m.source === "joyX" ? this.joy.x : this.joy.y) * m.gain;
   720→   720→          if (m.target === "cmd_x") vx += v;
   721→   721→          else if (m.target === "cmd_y") vy += v;
   722→   722→          else if (m.target === "cmd_yaw") wz -= v; // Stick rechts = Rechtsdrehung
   723→   723→        }
   724→   724→      }
   725→   725→      engine.cmd[0] = clamp(vx, VEL_BACK, VEL_FWD);
   726→   726→      engine.cmd[1] = clamp(vy, -VEL_LAT, VEL_LAT);
   727→   727→      engine.cmd[2] = clamp(wz, -VEL_ANG, VEL_ANG);
   728→   728→    }
   729→   729→
   730→   730→    // Button-Trigger (Edge)
   731→   731→    if (this.pendingPolicyId) {
   732→   732→      this.overridePolicy = this.pendingPolicyId;
   733→   733→      this.pendingPolicyId = null;
   734→   734→    }
   735→   735→    if (this.pendingPoseId) {
   736→   736→      const pose = poseById("microduck", this.pendingPoseId);
   737→   737→      this.pendingPoseId = null;
   738→   738→      if (pose) this.startDuckPose(pose.targets);
   739→   739→    }
   740→   740→
   741→   741→    // v2.1: Imitation manuell anwenden (Zielpose direkt als ctrl)
   742→   742→    if (this.mode === "manuell" && this.imitManual && this.imitClip && this.imitPlaying && this.imitTargetBuf) {
   743→   743→      engine.applyCtrlFromPose(this.imitTargetBuf);
   744→   744→      engine.stepPhysics();
   745→   745→      return;
   746→   746→    }
   747→   747→
   748→   748→    // Pose-Hold (kurze Overlay-Phase, dann zurück zur Policy)
   749→   749→    if (this.duckPoseHold) {
   750→   750→      const hold = this.duckPoseHold;
   751→   751→      const now = performance.now();
   752→   752→      const u = clamp((now - hold.t0) / hold.blendMs, 0, 1);
   753→   753→      const ctrl = engine.data.ctrl as Float32Array;
   754→   754→      for (let j = 0; j < meta.actionDim; j++) {
   755→   755→        ctrl[j] = hold.from[j] + (hold.to[j] - hold.from[j]) * u;
   756→   756→      }
   757→   757→      engine.stepPhysics();
   758→   758→      if (now - hold.t0 > hold.blendMs + hold.holdMs) {
   759→   759→        this.duckPoseHold = null;
   760→   760→        engine.lastAction.fill(0);
   761→   761→      }
   762→   762→      return;
   763→   763→    }
   764→   764→
   765→   765→    // Policy wählen
   766→   766→    let policy: OnnxPolicy | MlpPolicy | null = null;
   767→   767→    if (this.source === "es" && this.esView) {
   768→   768→      policy = this.esView;
   769→   769→    } else {
   770→   770→      const id = this.overridePolicy ?? this.selectedOnnx;
   771→   771→      policy = (await this.getOnnxSession(id)) ?? (await this.getOnnxSession(meta.defaultOnnx));
   772→   772→      if (!engine.data) return; // Modellwechsel während des Session-Ladens
   773→   773→    }
   774→   774→    if (!policy) {
   775→   775→      engine.stepPhysics();
   776→   776→      return;
   777→   777→    }
   778→   778→
   779→   779→    // ONNX braucht exakte Basis-Dimension; ES-MLP ggf. +2 (Imitations-Phase)
   780→   780→    const obs = policy instanceof MlpPolicy
   781→   781→      ? engine.obsFor(this.esObsExtra >= 2)
   782→   782→      : engine.obsFor(false);
   783→   783→    const act = policy instanceof MlpPolicy
   784→   784→      ? policy.forward(obs, this.actBuf)
   785→   785→      : await policy.forward(obs);
   786→   786→    this.actBuf.fill(0);
   787→   787→    this.actBuf.set(act.subarray(0, Math.min(act.length, this.actBuf.length)));
   788→   788→    engine.lastAction.set(this.actBuf);
   789→   789→    engine.stepWithAction(this.actBuf);
   790→   790→
   791→   791→    if (this.mode === "test") {
   792→   792→      this.testReward += stepRewardValue(engine, this.rewardCfg!, this.actBuf, this.prevAct);
   793→   793→      this.prevAct.set(this.actBuf);
   794→   794→    }
   795→   795→  }
   796→   796→
   797→   797→  private startDuckPose(targets: Record<string, number>) {
   798→   798→    const meta = this.engine.meta;
   799→   799→    const from = new Float32Array(this.engine.data.ctrl as Float32Array);
   800→   800→    const to = new Float32Array(meta.actionDim);
   801→   801→    for (let j = 0; j < meta.actionDim; j++) {
   802→   802→      const name = meta.jointNames[j];
   803→   803→      to[j] = targets[name] ?? meta.defaultPose[j];
   804→   804→    }
   805→   805→    this.duckPoseHold = { from, to, t0: performance.now(), blendMs: 400, holdMs: 1600 };
   806→   806→  }
   807→   807→
   808→   808→  // ── G1-Step (direkte Gelenk-Targets / ES-Policy) ────────────────────────────
   809→   809→
   810→   810→  private stepG1(): void {
   811→   811→    const engine = this.engine;
   812→   812→    const meta = engine.meta;
   813→   813→    const dim = meta.actionDim;
   814→   814→
   815→   815→    // Pose-Trigger (Edge)
   816→   816→    if (this.pendingPoseId) {
   817→   817→      const pose = poseById("unitree_g1", this.pendingPoseId);
   818→   818→      this.pendingPoseId = null;
   819→   819→      if (pose) this.startG1Blend(pose.targets);
   820→   820→    }
   821→   821→
   822→   822→    // Basis-Blending (0.4 s)
   823→   823→    if (this.g1Blend && this.g1Base) {
   824→   824→      const b = this.g1Blend;
   825→   825→      const u = clamp((performance.now() - b.t0) / b.dur, 0, 1);
   826→   826→      const e = u < 0.5 ? 2 * u * u : 1 - Math.pow(-2 * u + 2, 2) / 2;
   827→   827→      for (let j = 0; j < dim; j++) this.g1Base[j] = b.from[j] + (b.to[j] - b.from[j]) * e;
   828→   828→      if (u >= 1) this.g1Blend = null;
   829→   829→    }
   830→   830→
   831→   831→    if (this.mode === "manuell") {
   832→   832→      // Joystick-Offsets auf gemappten Gelenken
   833→   833→      const targets = new Float32Array(dim);
   834→   834→      targets.set(this.g1Base ?? engine.standPose);
   835→   835→      for (const m of this.mappings) {
   836→   836→        if (m.targetType !== "gelenk") continue;
   837→   837→        const v = (m.source === "joyX" ? this.joy.x : this.joy.y) * m.gain;
   838→   838→        for (const jn of m.target.split(",").map((s) => s.trim())) {
   839→   839→          const idx = meta.jointNames.indexOf(jn);
   840→   840→          if (idx >= 0) targets[idx] += v;
   841→   841→        }
   842→   842→      }
   843→   843→      engine.applyCtrlFromPose(targets);
   844→   844→      engine.stepPhysics();
   845→   845→      return;
   846→   846→    }
   847→   847→
   848→   848→    // v2.1: Punkt-Modus – cmd Richtung Punkt (für ES-Obs)
   849→   849→    if (this.pointMode && engine.targetPoint) {
   850→   850→      const [tx, ty] = engine.targetPoint;
   851→   851→      const p = engine.torsoPos();
   852→   852→      const dx = tx - p[0], dy = ty - p[1];
   853→   853→      const dist = Math.hypot(dx, dy);
   854→   854→      const yaw = engine.yaw();
   855→   855→      const desired = Math.atan2(dy, dx);
   856→   856→      let yerr = desired - yaw;
   857→   857→      while (yerr > Math.PI) yerr -= 2 * Math.PI;
   858→   858→      while (yerr < -Math.PI) yerr += 2 * Math.PI;
   859→   859→      engine.cmd[0] = dist > 0.15 ? 0.35 * Math.max(0, Math.cos(yerr)) : 0;
   860→   860→      engine.cmd[2] = clamp(1.2 * yerr, -1, 1);
   861→   861→    }
   862→   862→
   863→   863→    // Training/Test: ES-Policy (G1 hat keine ONNX-Policies)
   864→   864→    const mlp = this.esView;
   865→   865→    if (!mlp || !this.g1Base) {
   866→   866→      // noch nichts trainiert: Stand-Pose halten, Physik läuft weiter
   867→   867→      engine.applyCtrlFromPose(engine.standPose);
   868→   868→      engine.stepPhysics();
   869→   869→      return;
   870→   870→    }
   871→   871→    const obs = engine.obsFor(this.esObsExtra >= 2);
   872→   872→    const act = mlp.forward(obs, this.actBuf);
   873→   873→    engine.lastAction.set(act);
   874→   874→    engine.stepWithAction(act);
   875→   875→    if (this.mode === "test") {
   876→   876→      this.testReward += stepRewardValue(engine, this.rewardCfg!, act, this.prevAct);
   877→   877→      this.prevAct.set(act);
   878→   878→    }
   879→   879→  }
   880→   880→
   881→   881→  private startG1Blend(targets: Record<string, number>) {
   882→   882→    const meta = this.engine.meta;
   883→   883→    const to = new Float32Array(meta.actionDim);
   884→   884→    for (let j = 0; j < meta.actionDim; j++) {
   885→   885→      to[j] = targets[meta.jointNames[j]] ?? this.engine.standPose[j];
   886→   886→    }
   887→   887→    this.g1Blend = { from: (this.g1Base ?? this.engine.standPose).slice(), to, t0: performance.now(), dur: 400 };
   888→   888→  }

   890→   890→  // ── Auto-Recovery (sanftes Blenden zur Stand-Pose, kein Teleport) ───────────
   891→   891→
   892→   892→  private startRecovery() {
   893→   893→    this.recovering = true;
   894→   894→    this.recoverySteps = 0;
   895→   895→    this.recoveryUpright = 0;
   896→   896→    this.fallDebounce = 0;
   897→   897→    this.engine.lastAction.fill(0);
   898→   898→    this.emit();
   899→   899→  }
   900→   900→
   901→   901→  private stepRecovery() {
   902→   902→    const engine = this.engine;
   903→   903→    const meta = engine.meta;
   904→   904→    const stand = meta.id === "microduck" ? Float32Array.from(meta.defaultPose) : engine.standPose;
   905→   905→    const ctrl = engine.data.ctrl as Float32Array;
   906→   906→    for (let j = 0; j < meta.actionDim; j++) {
   907→   907→      ctrl[j] += (stand[j] - ctrl[j]) * 0.06; // sanft, pro Policy-Step
   908→   908→    }
   909→   909→    engine.stepPhysics();
   910→   910→    this.recoverySteps++;
   911→   911→    const upright = engine.projGravZ() < -0.85 && !engine.isFallen();
   912→   912→    this.recoveryUpright = upright ? this.recoveryUpright + 1 : 0;
   913→   913→    if (this.recoveryUpright >= 50) {
   914→   914→      // 1 s aufrecht → Policy übernimmt wieder (Zustand läuft weiter)
   915→   915→      this.recovering = false;
   916→   916→      this.recoveryUpright = 0;
   917→   917→      engine.lastAction.fill(0);
   918→   918→      this.prevAct.fill(0);
   919→   919→      this.emit();
   920→   920→    }
   921→   921→  }

   923→   923→  // ── Render-Loop (rAF) ───────────────────────────────────────────────────────
   924→   924→
   925→   925→  private renderLoop = () => {
   926→   926→    if (!this.running || !this.world) return;
   927→   927→    const now = performance.now();
   928→   928→    const dt = Math.min(0.05, (now - this.lastFrameT) / 1000 || 0.016);
   929→   929→    this.lastFrameT = now;
   930→   930→    const engine = this.engine;
   931→   931→    if (engine.data && engine.meta) {
   932→   932→      const [x, y, z] = engine.torsoPos();
   933→   933→      this.world.setCameraTarget(x, z, -y);
   934→   934→      if (engine.meta.id === "microduck") {
   935→   935→        syncDuck(this.world, engine);
   936→   936→      } else {
   937→   937→        syncG1(this.world, engine, engine.mujoco);
   938→   938→      }
   939→   939→      syncBall(this.world, engine);
   940→   940→      this.world.updateCamera(dt);
   941→   941→      this.world.renderer.render(this.world.scene, this.world.camera);
   942→   942→    }
   943→   943→    if (now - this.lastEmit > 250) {
   944→   944→      this.lastEmit = now;
   945→   945→      this.emit();
   946→   946→    }
   947→   947→    requestAnimationFrame(this.renderLoop);
   948→   948→  };

  1003→  1003→      webkitFullscreenElement?: Element;

  1005→  1005→    };

  1007→  1007→    const active = document.fullscreenElement ?? doc.webkitFullscreenElement;
  1008→  1008→    try {
  1009→  1009→      if (active) {
  1010→  1010→        if (document.exitFullscreen) await document.exitFullscreen();
  1011→  1011→        else await doc.webkitExitFullscreen?.();
  1012→  1012→        return;
  1013→  1013→      }
  1014→  1014→      if (el.requestFullscreen) await el.requestFullscreen();
  1015→  1015→      else if (full.webkitRequestFullscreen) await full.webkitRequestFullscreen();
  1016→  1016→      const orientation = screen.orientation as ScreenOrientation & {
  1017→  1017→        lock?: (o: string) => Promise<void>;
  1018→  1018→      };
  1019→  1019→      await orientation?.lock?.("landscape")?.catch?.(() => {});
  1020→  1020→    } catch {
